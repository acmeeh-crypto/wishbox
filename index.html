연동하지 않아도 현재 브라우저(로컬스토리지)에 소원이 안전하게 저장됩니다.
📌 Supabase 테이블 생성 SQL 보기 (처음 1회 실행 필요) 
-- Supabase SQL Editor에 복사해서 실행(Run)하세요:
CREATE TABLE IF NOT EXISTS wishes (
  id TEXT PRIMARY KEY,
  author_id TEXT NOT NULL,
  author_name TEXT NOT NULL,
  author_avatar TEXT NOT NULL,
  category TEXT NOT NULL,
  content TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'registered', -- acquired, registered, completed, expired
  created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now()),
  acquired_at TIMESTAMPTZ,
  registered_at TIMESTAMPTZ,
  expires_at TIMESTAMPTZ NOT NULL,
  completed_at TIMESTAMPTZ
);

-- 기존 테이블이 이미 있을 경우 새 컬럼 추가
ALTER TABLE wishes ADD COLUMN IF NOT EXISTS acquired_at TIMESTAMPTZ;
ALTER TABLE wishes ADD COLUMN IF NOT EXISTS registered_at TIMESTAMPTZ;

-- 누구나 읽고 쓸 수 있는 RLS 정책
ALTER TABLE wishes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read-write" ON wishes FOR ALL USING (true) WITH CHECK (true);

-- 실시간 업데이트 활성화
ALTER PUBLICATION supabase_realtime ADD TABLE wishes;

-- 미니게임(우주 탐험) 최고기록 공유 테이블
CREATE TABLE IF NOT EXISTS space_scores (
  player_id TEXT PRIMARY KEY, -- 'user1' 또는 'user2'
  best_score INT NOT NULL DEFAULT 0,
  best_stage INT NOT NULL DEFAULT 1,
  updated_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);
ALTER TABLE space_scores ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read-write" ON space_scores FOR ALL USING (true) WITH CHECK (true);
