두 사람이 서로 다른 스마트폰이나 컴퓨터에서 같은 소원 목록을 실시간으로 함께 보려면 무료 Supabase 프로젝트를 연동하세요!

연동하지 않아도 현재 브라우저(로컬스토리지)에 소원이 안전하게 저장됩니다.
📌 Supabase 테이블 생성 SQL 보기 (처음 1회 실행 필요) 
-- Supabase SQL Editor에 복사해서 실행(Run)하세요:
CREATE TABLE IF NOT EXISTS wishes (
  id TEXT PRIMARY KEY,
  author_id TEXT NOT NULL,
  author_name TEXT NOT NULL,
  author_avatar TEXT NOT NULL,
  category TEXT NOT NULL,
  content TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'registered', -- registered, completed, expired
  created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now()),
  expires_at TIMESTAMPTZ NOT NULL,
  completed_at TIMESTAMPTZ
);

-- 누구나 읽고 쓸 수 있는 RLS 정책
ALTER TABLE wishes ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read-write" ON wishes FOR ALL USING (true) WITH CHECK (true);

-- 실시간 업데이트 활성화
ALTER PUBLICATION supabase_realtime ADD TABLE wishes;
